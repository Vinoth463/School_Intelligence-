        WITH examination_performance AS (
            SELECT
                s.id AS student_id,
                s.admission_number,
                s.first_name,
                s.last_name,

                e.id AS examination_id,
                e.name AS examination_name,
                e.academic_period_id,
                e.exam_date,

                m.class_id,

                SUM(m.obtained_marks) AS total_obtained_marks,
                SUM(m.max_marks) AS total_max_marks,

                ROUND(
                    (
                        SUM(m.obtained_marks)
                        / NULLIF(SUM(m.max_marks), 0)
                    )::numeric * 100,
                    2
                ) AS overall_percentage

            FROM students s

            JOIN marks m
                ON m.student_id = s.id

            JOIN examinations e
                ON e.id = m.examination_id

            JOIN subjects sub
                ON sub.id = m.subject_id

            JOIN classes c
                ON c.id = m.class_id

            WHERE s.institution_id = %(institution_id)s AND e.institution_id = %(institution_id)s AND sub.institution_id = %(institution_id)s AND c.institution_id = %(institution_id)s AND e.academic_period_id = c.academic_period_id AND s.id = m.student_id AND e.academic_period_id = %(academic_period_id)s

            GROUP BY
                s.id,
                s.admission_number,
                s.first_name,
                s.last_name,
                e.id,
                e.name,
                e.academic_period_id,
                e.exam_date,
                m.class_id
        ),

        performance_changes AS (
            SELECT
                *,

                LAG(examination_id) OVER (
                    PARTITION BY student_id, academic_period_id
                    ORDER BY exam_date, examination_id
                ) AS previous_examination_id,

                LAG(examination_name) OVER (
                    PARTITION BY student_id, academic_period_id
                    ORDER BY exam_date, examination_id
                ) AS previous_examination_name,

                LAG(overall_percentage) OVER (
                    PARTITION BY student_id, academic_period_id
                    ORDER BY exam_date, examination_id
                ) AS previous_percentage

            FROM examination_performance
        )

        SELECT
            student_id,
            admission_number,
            first_name,
            last_name,

            academic_period_id,
            class_id,

            previous_examination_id,
            previous_examination_name,
            examination_id,
            examination_name,

            previous_percentage,
            overall_percentage AS current_percentage,

            ROUND(
                (overall_percentage - previous_percentage)::numeric,
                2
            ) AS percentage_change,

            CASE
                WHEN overall_percentage < previous_percentage
                    THEN 'declining'
                WHEN overall_percentage > previous_percentage
                    THEN 'improving'
                ELSE 'unchanged'
            END AS performance_status

        FROM performance_changes

        WHERE previous_percentage IS NOT NULL
            AND overall_percentage < previous_percentage
            AND previous_percentage - overall_percentage
                >= %(decline_threshold)s

        ORDER BY
            percentage_change ASC,
            student_id ASC
